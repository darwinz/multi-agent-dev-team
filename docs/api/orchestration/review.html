 49class QATestSuite(_Strict):
 50    """QA's LLM output: a pytest module plus a manifest mapping each test to a criterion."""
 51
 52    content: str
 53    tests: list[SuiteTest]
 54
 55    def check(self, n_criteria: int) -> list[str]:
 56        """Problems with the suite in words (empty means valid). A method, not a validator,
 57        so the QA retry can feed the problems back to the model."""
 58        try:
 59            tree = ast.parse(self.content)
 60        except SyntaxError as e:
 61            return [f"the test module has a syntax error: {e.msg} (line {e.lineno})"]
 62        defined = [
 63            n.name for n in tree.body
 64            if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef)) and n.name.startswith("test_")
 65        ]
 66        problems: list[str] = [
 67            f"class {n.name} holds test methods; tests must be plain top-level functions"
 68            for n in tree.body
 69            if isinstance(n, ast.ClassDef) and n.name.startswith("Test") and any(
 70                isinstance(m, (ast.FunctionDef, ast.AsyncFunctionDef)) and m.name.startswith("test")
 71                for m in n.body)
 72        ]
 73        seen: set[str] = set()
 74        for t in self.tests:
 75            if t.name in seen:
 76                problems.append(f"duplicate test name in the manifest: {t.name}")
 77            seen.add(t.name)
 78            if t.name not in defined:
 79                problems.append(f"manifest test {t.name} is not a top-level def test_* in the module")
 80            if not 1 <= t.criterion <= n_criteria:
 81                problems.append(f"test {t.name} maps to criterion {t.criterion}, but criteria are 1..{n_criteria}")
 82        mapped = {t.name for t in self.tests}
 83        problems += [f"test {name} is not in the manifest; map it to a criterion or remove it"
 84                     for name in dict.fromkeys(defined) if name not in mapped]
 85        covered = {t.criterion for t in self.tests}
 86        problems += [f"criterion {i} has no test" for i in range(1, n_criteria + 1) if i not in covered]
 87        return problems

 99class QAReport(_Strict):
100    """One QA round for a task: pytest counts and outcomes, issues, tampering and approval (computed in code)."""
101
102    task_id: str
103    round: int
104    test_file: str
105    suite_sha: str
106    suites_run: dict[str, str] = Field(description="task_id -> suite sha: this task's plus earlier approved tasks'")
107    tested_files: dict[str, str]
108    tests_run: int
109    passed: int
110    failed: int
111    errors: int
112    skipped: int
113    missing: list[str] = Field(description="manifest tests with no passing-or-failing result")
114    exit_code: int | None
115    timed_out: bool
116    duration_s: float
117    results: list[TestOutcome]
118    issues: list[Issue]
119    tampered: bool
120    qa_error: str | None
121    test_corrections: list[TestCorrection] = Field(default_factory=list,
122                                                   description="round 1 only: QA tests replaced before the rerun")
123    approved: bool
124
125    @model_validator(mode="after")
126    def _approval_is_earned(self) -> QAReport:
127        """Approval is computed in code from test results and never taken from a model."""
128        if not self.approved:
129            return self
130        problems = [
131            msg for ok, msg in [
132                (self.tests_run >= 1, "no tests ran"),
133                (self.passed == self.tests_run, "not every test passed"),
134                (self.failed == 0 and self.errors == 0 and self.skipped == 0, "tests failed, errored or were skipped"),
135                (not self.missing, "manifest tests are missing"),
136                (self.exit_code == 0, "pytest exit code is not 0"),
137                (not self.timed_out, "the run timed out"),
138                (self.qa_error is None, "QA reported an error"),
139                (not any(i.severity == "High" for i in self.issues), "a High issue is open"),
140            ] if not ok
141        ]
142        if problems:
143            raise ValueError("cannot approve: " + "; ".join(problems))
144        return self
