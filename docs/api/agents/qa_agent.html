 89def _clip(text: str, limit: int = MAX_ASSERTION) -> str:
 90    text = text.strip()
 91    return text if len(text) <= limit else text[: limit - 3] + "..."

 94def _assertion(case: CaseResult) -> str:
 95    """The failing source line plus pytest's E lines; else the message; else the text's tail."""
 96    lines = case.text.splitlines()
 97    source = [ln[1:].strip() for ln in lines if ln.startswith(">")]
 98    errors = [ln[1:].strip() for ln in lines if ln.startswith("E ")]
 99    if errors:
100        return _clip("\n".join(source[-1:] + errors))
101    return _clip(case.message or case.text[-MAX_ASSERTION:] or case.outcome)

104def _location(text: str, fallback: str) -> str:
105    """The last file.py:line in the failure text that isn't a test_* file."""
106    hits = [f"{m[1]}:{m[2]}" for m in _LOCATION.finditer(text) if not Path(m[1]).name.startswith("test_")]
107    return hits[-1] if hits else fallback

110def _replace_tests(content: str, replacements: dict[str, str]) -> str | None:
111    """Swap whole top-level test functions by name; every other line of the module is kept as is.
112    None if a name is not defined in the module or a replacement is not exactly one function of that name."""
113    funcs = (ast.FunctionDef, ast.AsyncFunctionDef)
114    try:
115        tree = ast.parse(content)
116    except SyntaxError:
117        return None
118    spans = {n.name: (min([n.lineno, *(d.lineno for d in n.decorator_list)]) - 1, n.end_lineno)
119             for n in tree.body if isinstance(n, funcs) and n.name in replacements}
120    if set(spans) != set(replacements):
121        return None
122    for name, source in replacements.items():
123        try:
124            body = ast.parse(source).body
125        except SyntaxError:
126            return None
127        if len(body) != 1 or not isinstance(body[0], funcs) or body[0].name != name:
128            return None
129    lines = content.splitlines(keepends=True)
130    for name, (start, end) in sorted(spans.items(), key=lambda kv: -kv[1][0]):
131        lines[start:end] = [replacements[name].strip("\n") + "\n"]
132    return "".join(lines)

135def _approved(r: dict[str, Any]) -> bool:
136    """The QAReport validator's conditions, computed rather than asserted."""
137    return (
138        r["tests_run"] >= 1 and r["passed"] == r["tests_run"] and r["failed"] == r["errors"] == r["skipped"] == 0
139        and not r["missing"] and r["exit_code"] == 0 and not r["timed_out"] and r["qa_error"] is None
140        and not any(i.severity == "High" for i in r["issues"])
141    )

157class _Reconciler:
158    """Matches one pytest run's JUnit cases to the suites' manifests, collecting outcomes and High issues."""
159
160    def __init__(self, suites: dict[str, QATestSuite], task_id: str, criteria: dict[str, list[str]]) -> None:
161        self.suites = suites
162        self.criteria = criteria
163        self.run = _Run()
164        self.stems = {qa_test_file(t)[:-3]: t for t in suites}
165        # issues from other tasks' suites (the regression check) say which task regressed
166        self.prefix = {t: "" if t == task_id else f"regression in {t}: " for t in suites}
167
168    def issue(self, owner: str | None, test: str | None, criterion: str | None, location: str,
169              assertion: str) -> None:
170        """Record one High issue, prefixed with its owning task when that is another task's suite."""
171        text = _clip(self.prefix.get(owner, "") + assertion) if owner else _clip(assertion)
172        self.run.issues.append(Issue(severity="High", test=test, criterion=criterion, location=location,
173                                     assertion=text, fix=f"Make {test or location} pass: {text}"))
174
175    def criterion_of(self, owner: str, test: str) -> str | None:
176        """The acceptance-criterion text a manifest test covers, if the manifest names one."""
177        n = next((t.criterion for t in self.suites[owner].tests if t.name == test), None)
178        texts = self.criteria.get(owner, [])
179        if n is None:
180            return None
181        return texts[n - 1] if 0 < n <= len(texts) else f"{owner} criterion {n}"
182
183    def _owner(self, case: CaseResult) -> str | None:
184        """The task whose suite module a JUnit case came from, if any."""
185        return next((t for stem, t in self.stems.items()
186                     if case.classname == stem or case.classname.endswith("." + stem)
187                     or (not case.classname and case.name == stem)), None)
188
189    def record_case(self, case: CaseResult) -> tuple[str, str] | None:
190        """Record one JUnit case; returns (owner, test) when it is a manifest test, else None."""
191        owner = self._owner(case)
192        test = case.name.split("[")[0]
193        node = f"{case.classname}::{case.name}" if case.classname else case.name
194        self.run.results.append(TestOutcome(name=node, outcome=case.outcome, message=_clip(case.message)))
195        if owner is not None and any(t.name == test for t in self.suites[owner].tests):
196            if case.outcome != "passed":
197                what = "skipped: " if case.outcome == "skipped" else ""
198                self.issue(owner, test, self.criterion_of(owner, test), _location(case.text, test),
199                           what + _assertion(case))
200            return owner, test
201        if owner is not None and not case.classname:  # the module failed to import or collect
202            self.issue(owner, None, None, _location(case.text, qa_test_file(owner)), _assertion(case))
203        else:
204            self.issue(owner, case.name or None, None, _location(case.text, node or "QA test run"),
205                       f"unexpected test {node or '(unnamed)'} ({case.outcome}), not in any QA suite manifest: "
206                       + _assertion(case))
207        return None
208
209    def flag_missing(self, seen: set[tuple[str, str]], *, report_issues: bool) -> None:
210        """Every manifest test with no JUnit result is missing, which still blocks approval."""
211        for owner, suite in self.suites.items():
212            for t in suite.tests:
213                if (owner, t.name) not in seen:
214                    self.run.missing.append(t.name)
215                    if report_issues:
216                        self.issue(owner, t.name, self.criterion_of(owner, t.name), t.name,
217                                   f"{t.name} did not run (no test result for it)")

220def _later_tasks(later: Sequence[Task]) -> str:
221    """The suite prompt's later-tasks section, or nothing for the last task."""
222    if not later:
223        return ""
224    lines = ["Later tasks (built after this one; do not test behaviour they define or change):"]
225    for t in later:
226        lines += [f"{t.task_id}: {t.title}", *(f"  - {c}" for c in t.acceptance_criteria)]
227    return "\n".join(lines) + "\n\n"

237def _explain_uncovered(problems: list[str], criteria: Sequence[str]) -> list[str]:
238    """Quote each uncovered criterion in the retry ask: a bare number gave the model nothing to act on
239    (live BST runs: a "tests pass with pytest" criterion stayed unmapped on both attempts)."""
240    out = list(problems)
241    uncovered = False
242    for n, text in enumerate(criteria, 1):
243        bare = f"criterion {n} has no test"
244        if bare in out:
245            out[out.index(bare)] = f"criterion {n} ({' '.join(text.split())!r}) has no test"
246            uncovered = True
247    return [*out, _TEST_CRITERION_HINT] if uncovered else out

250class QAAgent:
251    """The QA agent: writes the pytest suite for a task, runs it, and turns the results into a QAReport."""
252
253    def __init__(self, settings: Settings | None = None, *, model: BaseChatModel | None = None) -> None:
254        """Settings, plus an optional model (wrapped with retry, breaker and cache); else built on first use."""
255        self.settings = settings or Settings.from_env()
256        self.model = resilient(model, self.settings, "qa") if model is not None else None
257
258    def _structured(self, schema: type) -> Any:
259        if self.model is None:
260            self.model = make_model(self.settings, "qa")
261        # suite, fix plan and audit are side-effect free and run-stable: response-cache eligible
262        return self.model.cached().with_structured_output(schema, include_raw=True)
263
264    @property
265    def _ws(self) -> Workspace:
266        return Workspace(self.settings.workspace_dir)

268    @property
269    def _qa_dir(self) -> Path:
270        return self.settings.state_dir / "qa"

274    def generate_suite(self, handoff: TaskHandoff, files: dict[str, str], later: Sequence[Task] = ()) -> QATestSuite:
275        """Ask for a suite; on an invalid one retry once with the problems appended. `later` are the
276        tasks built after this one: QA sees them so its frozen suite does not pin behaviour they change."""
277        try:
278            structured = self._structured(QATestSuite)
279        except Exception as e:  # missing provider package, bad model id
280            raise QAError(f"QA model setup failed: {type(e).__name__}: {e}") from e
281        criteria = "\n".join(f"{i}. {c}" for i, c in enumerate(handoff.acceptance_criteria, 1))
282        code = "\n\n".join(f"--- {path} ---\n{text}" for path, text in sorted(files.items())) or "(no files)"
283        messages: list[Any] = [
284            SystemMessage(QA_TESTS_PROMPT),
285            HumanMessage(f"Task {handoff.task_id}: {handoff.title}\n{handoff.description}\n\n"
286                         f"Acceptance criteria:\n{criteria}\n\n{_later_tasks(later)}Delivered files:\n{code}"),
287        ]
288        problems: list[str] = []
289        for attempt in range(2):
290            try:
291                out = structured.invoke(messages)
292            except Exception as e:
293                raise QAError(f"QA model call failed: {type(e).__name__}: {e}") from e
294            suite = out.get("parsed")
295            if cut_off(out.get("raw")):  # the parse error of a truncated reply would mislead the retry
296                problems = [("your reply hit the output-token limit and was cut off before it was complete. "
297                            "Write a shorter suite: fewer, compact tests (pytest.mark.parametrize for "
298                            "similar cases), no long docstrings or comments")]
299            elif out.get("parsing_error") is not None or suite is None:
300                problems = [str(out.get("parsing_error") or "the model returned no structured suite")]
301            else:
302                problems = suite.check(len(handoff.acceptance_criteria))
303                if not problems:
304                    return suite
305                problems = _explain_uncovered(problems, handoff.acceptance_criteria)
306            if attempt == 0:
307                messages.append(HumanMessage(
308                    "Your test suite was invalid:\n- " + "\n- ".join(problems)
309                    + "\n\nReturn a corrected suite that fixes these problems."))
310        raise QAError("QA produced an invalid test suite twice: " + "; ".join(problems))

314    def _restore(self, suites: dict[str, QATestSuite]) -> tuple[dict[str, str], bool]:
315        """Write each suite's state-held content to its canonical file and the workspace root.
316        Returns ({id: sha}, tampered): code the Coder runs could rewrite either copy on disk."""
317        ws, shas, tampered = self._ws, {}, False
318        for task_id, suite in suites.items():
319            content = suite.content
320            path = self._checked_write_args(qa_test_file(task_id), content)
321            canonical = canonical_suite_path(self.settings, task_id)
322            if not canonical.is_file() or canonical.read_text(encoding="utf-8", errors="replace") != content:
323                tampered = True
324                canonical.parent.mkdir(parents=True, exist_ok=True)
325                canonical.write_text(content, encoding="utf-8")
326            target = ws.resolve(path)
327            if not target.is_file() or target.read_text(encoding="utf-8", errors="replace") != content:
328                tampered = True
329            guarded_tool("write_file", lambda path=path, content=content: write_file(ws, path, content))
330            shas[task_id] = _sha(content)
331        return shas, tampered
332
333    def _checked_write_args(self, path: str, content: str) -> str:
334        """QA's direct write_file calls get the Coder's argument model plus the workspace check up front
335        (spec §5.7), so a bad path never reaches write_file. Raises ToolError (or ValidationError)."""
336        a = ARG_MODELS["write_file"].model_validate({"path": path, "content": content})
337        self._ws.resolve(a.path)  # WorkspaceError if it escapes the workspace
338        return a.path
339
340    def _pytest(self, files: list[str], junit: Path) -> ExecOutcome:
341        """One sandboxed pytest call over every suite file. exec_python's pytest mode takes a single
342        target, so a script snippet calls pytest.main with all of them under the same timeout."""
343        # QA's own empty ini: the Coder's pytest.ini / pyproject addopts must not reshape QA's run
344        ini = self._qa_dir / "pytest.ini"
345        ini.parent.mkdir(parents=True, exist_ok=True)
346        ini.write_text("[pytest]\n", encoding="utf-8")
347        args = ["-q", "-p", "no:cacheprovider", "-c", str(ini), f"--rootdir={self._ws.root}",
348                "--continue-on-collection-errors", f"--junitxml={junit}", *files]
349        code = f"import sys\nimport pytest\n\nsys.exit(int(pytest.main({json.dumps(args)})))\n"
350        mode, timeout = "script", self.settings.exec_timeout_s
351        if mode != "script" or not 0 < timeout <= self.settings.exec_timeout_s:
352            raise ToolError(f"QA exec_python needs mode 'script' and 0 < timeout <= {self.settings.exec_timeout_s:g}")
353        return guarded_tool("exec_python", lambda: exec_python(
354            self._ws, record_dir=self._qa_dir / "exec", code=code, mode=mode, timeout=timeout))
355
356    def run_suites(self, suites: dict[str, QATestSuite], *, task_id: str, round: int) -> QAReport:
357        """Restore every suite from canonical, run them in one pytest call, reconcile with the manifests.
358        Never calls the model; each issue's fix is the fallback text."""
359        if not suites:
360            raise ValueError("run_suites needs at least one suite")  # pytest would collect the whole workspace
361        return self._run(suites, task_id=task_id, round=round, criteria={})
362
363    def _run(self, suites: dict[str, QATestSuite], *, task_id: str, round: int,
364             criteria: dict[str, list[str]]) -> QAReport:
365        shas, tampered = self._restore(suites)
366        self._qa_dir.mkdir(parents=True, exist_ok=True)
367        junit = self._qa_dir / f"junit-{task_id.lower()}-r{round}.xml"
368        junit.unlink(missing_ok=True)
369        out = self._pytest([qa_test_file(t) for t in suites], junit)
370        run = self._reconcile(suites, task_id, criteria, out, junit)
371        count = {k: sum(r.outcome == k for r in run.results) for k in ("passed", "failed", "error", "skipped")}
372        return _report(
373            task_id=task_id, round=round, test_file=qa_test_file(task_id) if task_id in suites else "",
374            suite_sha=shas.get(task_id, ""), suites_run=shas, tested_files={},
375            tests_run=len(run.results), passed=count["passed"], failed=count["failed"], errors=count["error"],
376            skipped=count["skipped"], missing=run.missing, exit_code=out.exit_code, timed_out=out.timed_out,
377            duration_s=out.duration_s, results=run.results, issues=run.issues, tampered=tampered, qa_error=None,
378        )
379
380    def _reconcile(self, suites: dict[str, QATestSuite], task_id: str, criteria: dict[str, list[str]],
381                   out: ExecOutcome, junit: Path) -> _Run:
382        """Every manifest test of every suite must appear in the JUnit cases as passed (spec §6)."""
383        rec = _Reconciler(suites, task_id, criteria)
384        result = None if out.timed_out else parse_junit(junit)
385        tail = _clip((out.stdout + "\n" + out.stderr)[-_TAIL:], MAX_ASSERTION - 80)
386        if out.timed_out:
387            rec.issue(None, None, None, "QA test run",
388                      f"QA test run timed out after {self.settings.exec_timeout_s:g} s")
389        elif result is None:
390            rec.issue(None, None, None, "QA test run",
391                      f"QA could not read pytest's JUnit XML (exit {out.exit_code}): {tail}")
392        seen = {key for case in (result.cases if result else []) if (key := rec.record_case(case))}
393        # a timeout or unreadable XML already has its one High issue, so missing tests add none then
394        rec.flag_missing(seen, report_issues=result is not None)
395        failing = any(r.outcome in ("failed", "error") for r in rec.run.results)
396        if result is not None and out.exit_code not in (0, None) and not failing:
397            rec.issue(None, None, None, "QA test run", f"pytest exited {out.exit_code} with no failing test: {tail}")
398        return rec.run

402    def _fresh(self, request: ReviewRequest) -> tuple[dict[str, str], dict[str, str], list[Issue]]:
403        """Compare each requested file's on-disk sha with the request. Returns (shas, contents, issues)."""
404        ws, shas, contents, issues = self._ws, {}, {}, []
405        for path, sha in sorted(request.files.items()):
406            try:
407                target = ws.resolve(path)
408                disk = sha256_file(target) if target.is_file() else None
409            except (ToolError, OSError):  # outside the workspace, or a name the OS rejects
410                disk = None
411            if disk is not None:
412                shas[path] = disk
413                contents[path] = target.read_text(encoding="utf-8", errors="replace")
414            if disk != sha:
415                now = f"sha {disk[:12]}" if disk else "missing"
416                issues.append(Issue(
417                    severity="High", test=None, criterion=None, location=path,
418                    assertion=_clip(f"changed after review request: {path} (requested sha {sha[:12]}, on disk {now})"),
419                    fix=f"Send a new review request with the current contents of {path}"))
420        return shas, contents, issues

422    def _fixes(self, issues: list[Issue], contents: dict[str, str]) -> list[Issue]:
423        """One fix-plan call; its fixes replace the fallbacks by test name. Never raises."""
424        try:
425            code = "\n\n".join(f"--- {p} ---\n{t}" for p, t in sorted(contents.items())) or "(no files)"
426            failures = "\n".join(f"- {i.test or i.location}: {i.assertion}" for i in issues)
427            out = self._structured(FixPlan).invoke([
428                SystemMessage(QA_FIX_PROMPT), HumanMessage(f"Code:\n{code}\n\nFailures:\n{failures}")])
429            plan = out.get("parsed") if out.get("parsing_error") is None else None
430            by_test = {f.test: f.fix for f in plan.fixes if f.fix.strip()} if plan else {}
431        except Exception as e:  # noqa: BLE001 - keep the fallback fixes, but say so
432            log_event("qa_fix_plan_fallback", level=logging.WARNING, error_class=error_class(e), message=str(e))
433            return issues
434        return [i.model_copy(update={"fix": by_test[i.test]}) if i.test in by_test else i for i in issues]

436    def _save_suite(self, task_id: str, suite: QATestSuite) -> None:
437        """The canonical copy (replacing any left by an earlier run) and its workspace copy."""
438        path = self._checked_write_args(qa_test_file(task_id), suite.content)
439        canonical = canonical_suite_path(self.settings, task_id)
440        canonical.parent.mkdir(parents=True, exist_ok=True)
441        canonical.write_text(suite.content, encoding="utf-8")
442        guarded_tool("write_file", lambda: write_file(self._ws, path, suite.content))

444    def _check_tests(self, handoff: TaskHandoff, suite: QATestSuite, issues: list[Issue],
445                     contents: dict[str, str]) -> tuple[QATestSuite, list[TestCorrection]]:
446        """Round 1 only: ask QA whether its own failing tests contradict the criteria, and splice in the
447        replacements it gives. Any invalid replacement keeps the suite unchanged. Never raises."""
448        manifest = {t.name for t in suite.tests}
449        failing = sorted({i.test for i in issues
450                          if i.test in manifest and not i.assertion.startswith("regression in ")})
451        if not failing:
452            return suite, []
453        try:
454            tree = ast.parse(suite.content)
455            source = {n.name: ast.get_source_segment(suite.content, n) or "" for n in tree.body
